// Android-only push token registration.
//
// The Android app is a Capacitor shell that loads this web app f
// https://app.marvelcardvault.com, so this module is what actually asks the
// OS for notification permission and hands the FCM token to the
//
// Deliberately scoped to Android: iOS push is not wired up yet,
// on every other platform rather than half-register a device we can't send to.
// The backend accepts 'ios' too, so extending this later is a on

// Registration is triggered from onAuthStateChanged, which fires
// refresh and on appStateChange. These guards keep us from re-POSTing the same
// token on every one of those.
let registering = false;
let lastRegisteredToken: string | null = null;

async function sendTokenToBackend(token: string): Promise<void> {
  if (token === lastRegisteredToken) return;

  // Relative path on purpose: inside the shell the WebView origin already is
  // https://app.marvelcardvault.com, so this resolves to
  // https://app.marvelcardvault.com/api/push/register-token — while still
  // pointing at localhost during web development instead of hitt
  // apiRequest attaches the existing Firebase bearer/session headers.
  await apiRequest('POST', '/api/push/register-token', {
    token,
    platform: 'android',
  });

  lastRegisteredToken = token;
  console.log('[Push] FCM token registered');
}

  refreshUser: async () => {},

          firebaseUid: firebaseUser.uid,
          email: firebaseUser.email,
          displayName: firebaseUser.displayName,
          photoURL: firebaseUser.photoURL,
        }),
      });

      if (response.ok) {
        const data = await response.json();
        console.log('User synced with backend:', data.user);

        // Update app store with backend user data (including adm
        setCurrentUser({
          id: data.user.id,
          name: data.user.displayName || data.user.username,
          email: data.user.email,
          avatar: data.user.photoURL || '',
          isAdmin: data.user.isAdmin,
          plan: data.user.plan,
          subscriptionStatus: data.user.subscriptionStatus,
          onboardingComplete: data.user.onboardingComplete || false,
          username: data.user.username
        });
      } else {
        console.error('Failed to sync user with backend');
      }
    } catch (error) {
      console.error('Error syncing user:', error);
    }

      await syncUserWithBackend(user);
    }
  };

      handleRedirect();

        setUser(firebaseUser);

        if (firebaseUser) {
          await syncUserWithBackend(firebaseUser);

          // Android push: no-ops on web/iOS, and never throws.
